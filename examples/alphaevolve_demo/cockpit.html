def solve_knapsack(capacity: int, weights: list[int], values: list[int]) -> list[int]:
    """gen_000: Simple greedy selection by raw item value."""
    n = len(weights)
    order = sorted(range(n), key=lambda i: values[i], reverse=True)
    picked, rem = [], capacity
    for i in order:
        if weights[i] <= rem:
            picked.append(i); rem -= weights[i]
    return picked
